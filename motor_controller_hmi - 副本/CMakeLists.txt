cmake_minimum_required(VERSION 3.24)
project(MotorTune VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Sql Test Concurrent)
add_library(motortune_core src/protocol.cpp src/sampling.cpp src/backend.cpp src/session.cpp src/archive.cpp)
target_include_directories(motortune_core PUBLIC src)
target_link_libraries(motortune_core PUBLIC Qt6::Core Qt6::Sql)
target_compile_definitions(motortune_core PUBLIC MOTORTUNE_VERSION="${PROJECT_VERSION}")
if(MSVC)
  target_compile_options(motortune_core PUBLIC /utf-8 /W4)
else()
  target_compile_options(motortune_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(motortune_controls src/motion_control.cpp src/notice_bar.cpp)
target_link_libraries(motortune_controls PUBLIC motortune_core Qt6::Widgets)
add_executable(MotorTune WIN32 src/main.cpp src/window.cpp src/workflows.cpp src/scope.cpp)
target_link_libraries(MotorTune PRIVATE motortune_controls Qt6::Concurrent)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE motortune_core Qt6::Test)
add_test(NAME core_tests COMMAND core_tests)
add_executable(session_state_tests tests/session_state_tests.cpp)
target_link_libraries(session_state_tests PRIVATE motortune_core Qt6::Test)
add_test(NAME session_state_tests COMMAND session_state_tests)
if(WIN32)
  add_executable(can_read_probe EXCLUDE_FROM_ALL tests/can_read_probe.cpp)
  target_link_libraries(can_read_probe PRIVATE motortune_core)
  add_library(fake_can_driver SHARED tests/fake_can_driver.cpp)
  set_target_properties(fake_can_driver PROPERTIES PREFIX "" OUTPUT_NAME ControlCANFD
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/test-driver")
  add_executable(native_backend_tests tests/native_backend_tests.cpp)
  target_link_libraries(native_backend_tests PRIVATE motortune_core Qt6::Test)
  target_compile_definitions(native_backend_tests PRIVATE TEST_DRIVER_DIR="$<TARGET_FILE_DIR:fake_can_driver>")
  add_dependencies(native_backend_tests fake_can_driver)
  add_test(NAME native_backend_tests COMMAND native_backend_tests)
  set_tests_properties(native_backend_tests PROPERTIES TIMEOUT 15)
endif()
add_executable(operation_ui_tests tests/operation_ui_tests.cpp src/window.cpp src/workflows.cpp src/scope.cpp)
target_link_libraries(operation_ui_tests PRIVATE motortune_controls Qt6::Concurrent Qt6::Test)
add_test(NAME operation_ui_tests COMMAND operation_ui_tests)
set_tests_properties(operation_ui_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
add_executable(sampling_tests tests/sampling_tests.cpp)
target_link_libraries(sampling_tests PRIVATE motortune_core Qt6::Test)
add_test(NAME sampling_tests COMMAND sampling_tests)
add_executable(scope_benchmark tests/scope_benchmark.cpp src/scope.cpp)
target_link_libraries(scope_benchmark PRIVATE motortune_core Qt6::Widgets)
add_executable(motion_tests tests/motion_tests.cpp)
target_link_libraries(motion_tests PRIVATE motortune_controls Qt6::Test)
add_test(NAME motion_tests COMMAND motion_tests)
set_tests_properties(motion_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
add_test(NAME demo_smoke COMMAND MotorTune --smoke-test)
set_tests_properties(demo_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/compat/python/python.exe")
  add_test(NAME extension_tests COMMAND "${CMAKE_CURRENT_SOURCE_DIR}/compat/python/python.exe" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_extensions.py")
endif()
install(TARGETS MotorTune RUNTIME DESTINATION .)
install(DIRECTORY compat DESTINATION . PATTERN "*.zip" EXCLUDE PATTERN "__pycache__" EXCLUDE)
install(DIRECTORY library docs DESTINATION .)
install(FILES scripts/Install.ps1 scripts/Install.cmd DESTINATION .)
install(FILES README.md DESTINATION .)
qt_generate_deploy_app_script(TARGET MotorTune OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
install(CODE "set(QT_DEPLOY_BIN_DIR \".\")")
install(SCRIPT ${deploy_script})
if(WIN32 AND EXISTS "${Qt6_DIR}/../../../plugins/platforms/qoffscreen.dll")
  install(FILES "${Qt6_DIR}/../../../plugins/platforms/qoffscreen.dll" DESTINATION plugins/platforms)
endif()
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
include(CPack)
